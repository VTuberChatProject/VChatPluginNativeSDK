# C++ 插件示例使用与 Rust 生成器相同的 Protobuf 31.1。
cmake_minimum_required(VERSION 3.24)
project(vchat_cpp_plugin LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Protobuf 31.1.0 EXACT CONFIG REQUIRED)
if(NOT VCHAT_PROTOC)
  find_program(VCHAT_PROTOC protoc REQUIRED)
endif()
execute_process(COMMAND ${VCHAT_PROTOC} --version OUTPUT_VARIABLE VCHAT_PROTOC_VERSION OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT VCHAT_PROTOC_VERSION STREQUAL "libprotoc 31.1")
  message(FATAL_ERROR "VChat requires protoc 31.1; set VCHAT_PROTOC to the pinned executable")
endif()
add_library(vchat_cpp_example SHARED cpp_example.cpp ../Core/Interfaces/internal_plugin.proto)
protobuf_generate(TARGET vchat_cpp_example LANGUAGE cpp APPEND_PATH PROTOC_EXE ${VCHAT_PROTOC})
target_include_directories(vchat_cpp_example PRIVATE ${CMAKE_CURRENT_BINARY_DIR} ../Core/Interfaces)
target_link_libraries(vchat_cpp_example PRIVATE protobuf::libprotobuf)
target_compile_features(vchat_cpp_example PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(vchat_cpp_example PRIVATE /utf-8)
endif()
