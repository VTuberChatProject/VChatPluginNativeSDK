# 内部 C++ 插件仅依赖 C ABI 2 头文件，发行插件通过宿主注入函数表调用接口。
cmake_minimum_required(VERSION 3.24)
project(vchat_cpp_plugin LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT VCHAT_SDK)
  set(VCHAT_SDK "${CMAKE_CURRENT_LIST_DIR}/../Core/Interfaces")
endif()
find_path(VCHAT_INTERFACE_INCLUDE VCPluginInterface.h
  PATHS "${VCHAT_SDK}" "${VCHAT_SDK}/include" NO_DEFAULT_PATH REQUIRED)
add_library(vchat_cpp_example SHARED cpp_example.cpp)
target_include_directories(vchat_cpp_example PRIVATE "${VCHAT_INTERFACE_INCLUDE}")
target_compile_features(vchat_cpp_example PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(vchat_cpp_example PRIVATE /utf-8 /W4)
else()
  target_compile_options(vchat_cpp_example PRIVATE -Wall -Wextra -Werror)
endif()
